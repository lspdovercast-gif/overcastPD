-- ========================================================= -- LSPD ADMIN PANEL — kompletna SQL setup skripta -- Pokreni ovo u Supabase: SQL Editor -> New query -> Run -- ========================================================= -- ---------- ADMIN_PERMISSIONS (ko je admin/super admin) ---------- create table if not exists admin_permissions ( user_id uuid primary key references auth.users(id) on delete cascade, email text, fullname text, rank text, is_admin boolean default false, is_super_admin boolean default false, can_view_applications boolean default false ); alter table admin_permissions enable row level security; create policy "Ulogovani mogu čitati sve profile" on admin_permissions for select to authenticated using (true); create policy "Samo super admin mijenja dozvole" on admin_permissions for update to authenticated using ( exists (select 1 from admin_permissions ap where ap.user_id = auth.uid() and ap.is_super_admin = true) ); create policy "Samo super admin dodaje profile" on admin_permissions for insert to authenticated with check ( exists (select 1 from admin_permissions ap where ap.user_id = auth.uid() and ap.is_super_admin = true) ); -- ---------- APPLICATIONS (prijave) ---------- create table if not exists applications ( id bigint generated always as identity primary key, case_id text, fullname text, discord text, contact text, age text, experience text, motivation text, scenario text, status text default 'Na čekanju', created_at timestamptz default now() ); alter table applications enable row level security; create policy "Ulogovani sa dozvolom čitaju prijave" on applications for select to authenticated using ( exists (select 1 from admin_permissions ap where ap.user_id = auth.uid() and ap.can_view_applications = true) ); create policy "Admin mijenja/briše prijave" on applications for update to authenticated using ( exists (select 1 from admin_permissions ap where ap.user_id = auth.uid() and ap.is_admin = true) ); create policy "Admin briše prijave" on applications for delete to authenticated using ( exists (select 1 from admin_permissions ap where ap.user_id = auth.uid() and ap.is_admin = true) );
